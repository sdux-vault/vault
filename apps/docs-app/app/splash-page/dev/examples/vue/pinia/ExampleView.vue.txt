<script setup lang="ts">
import { storeToRefs } from 'pinia';
import { Employee } from './employee.model';
import { useEmployeeStore } from './employee.store';

const sample: Employee[] = [
  { id: 11, name: 'Luke' },
  { id: 38, name: 'Leia' },
  { id: 9, name: 'Han' }
];

const employeeStore = useEmployeeStore();
const { error, loading, value } = storeToRefs(employeeStore);

function loadSample(): void {
  employeeStore.replace(sample);
}

async function loadSampleAsync(): Promise<void> {
  await employeeStore.replaceAsync();
}

function resetState(): void {
  employeeStore.reset();
}
</script>

<template>
  <div>
    <div>
      <div v-if="loading">Loading...</div>
      <div v-else-if="error">{{ error }}</div>
      <textarea v-else readonly :value="JSON.stringify(value, null, 2)" />
    </div>

    <div>
      <button type="button" @click="loadSample">Load Sample State</button>

      <button type="button" @click="loadSampleAsync">Load Async State</button>

      <button type="button" @click="resetState">Reset State</button>
    </div>
  </div>
</template>
