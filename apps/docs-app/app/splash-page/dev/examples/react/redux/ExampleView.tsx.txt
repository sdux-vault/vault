import { Employee } from './employee.model';
import { useEmployeeFacade } from './useEmployeeFacade';

const sample: Employee[] = [
  { id: 11, name: 'Luke' },
  { id: 38, name: 'Leia' },
  { id: 9, name: 'Han' }
];

export function ExampleView() {
  const { state, replace, reset } = useEmployeeFacade();

  const loadSample = () => {
    replace(sample);
  };

  const resetState = () => {
    reset();
  };

  return (
    <div>
      <div>
        {state.loading ? (
          <div>Loading...</div>
        ) : state.error ? (
          <div>{state.error}</div>
        ) : (
          <textarea readOnly value={JSON.stringify(state.value, null, 2)} />
        )}
      </div>

      <div>
        <button type="button" onClick={loadSample}>
          Load Sample State
        </button>

        <button type="button" onClick={resetState}>
          Reset State
        </button>
      </div>
    </div>
  );
}
