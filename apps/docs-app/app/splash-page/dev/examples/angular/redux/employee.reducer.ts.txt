import { createReducer, on } from '@ngrx/store';
import { EmployeeActions } from './employee.actions';
import { Employee } from './employee.model';
import { EmployeeState, initialEmployeeState } from './employee.state';

export const employeeFeatureKey = 'employees';

const seededEmployees = [
  { id: 1, name: 'Ada' },
  { id: 2, name: 'Brian' }
];

const filterOddEmployees = (employees: Employee[]): Employee[] => {
  return employees.filter((employee) => employee.id % 2 !== 0);
};

const sortEmployeesByName = (employees: Employee[]): Employee[] => {
  const sortedEmployees = [...employees];

  sortedEmployees.sort((left, right) => left.name.localeCompare(right.name));

  return sortedEmployees;
};

const prepareEmployees = (employees: Employee[]): Employee[] => {
  return sortEmployeesByName(filterOddEmployees(employees));
};

export const employeeReducer = createReducer<EmployeeState>(
  initialEmployeeState,
  on(EmployeeActions.seed, (state) => ({
    ...state,
    loading: false,
    value: prepareEmployees(seededEmployees),
    error: null
  })),
  on(EmployeeActions.replace, (state, { employees }) => ({
    ...state,
    loading: false,
    value: prepareEmployees(employees),
    error: null
  })),
  on(EmployeeActions.reset, () => initialEmployeeState)
);
